cmake_minimum_required(VERSION 3.22)
project(palm_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
if(NOT EXISTS "${CLOUDPILOT_SOURCE}/src/uarm/Makefile")
    message(FATAL_ERROR "Set CLOUDPILOT_SOURCE to the patched Cloudpilot checkout")
endif()
if(NOT EXISTS "${SDL2_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set SDL2_SOURCE to SDL2 release-2.32.10")
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
# SDLActivity always initializes these JNI callbacks, even without HID devices.
set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
add_subdirectory("${SDL2_SOURCE}" "${CMAKE_BINARY_DIR}/sdl2")
set(core "${CLOUDPILOT_SOURCE}/src")
find_program(HOST_PYTHON python3 REQUIRED)
if(REMOTE_APP MATCHES "^novii")
    set(backends uarm)
else()
    set(backends cloudpilot)
endif()
foreach(backend IN LISTS backends)
    execute_process(COMMAND "${HOST_PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/core_sources.py" "${CLOUDPILOT_SOURCE}" "${backend}"
        OUTPUT_VARIABLE core_sources COMMAND_ERROR_IS_FATAL ANY OUTPUT_STRIP_TRAILING_WHITESPACE)
    string(REPLACE "\n" ";" core_sources "${core_sources}")
    execute_process(COMMAND "${HOST_PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/core_sources.py" "${CLOUDPILOT_SOURCE}" "${backend}" includes
        OUTPUT_VARIABLE core_includes COMMAND_ERROR_IS_FATAL ANY OUTPUT_STRIP_TRAILING_WHITESPACE)
    string(REPLACE "\n" ";" core_includes "${core_includes}")
    if(backend STREQUAL uarm)
        set(target ${REMOTE_APP})
        set(frontend "${core}/uarm/web/uarm.cpp")
    else()
        set(target omni)
        set(frontend "${core}/cloudpilot/web/Cloudpilot.cpp" native_assert.cpp)
    endif()
    add_library(${target} SHARED native_main.cpp remote_backend.cpp ${core_sources} ${frontend})
    target_include_directories(${target} PRIVATE ${core_includes} "${core}/${backend}/web")
    target_compile_definitions(${target} PRIVATE _FILE_OFFSET_BITS=64)
    if(REMOTE_APP STREQUAL novii_pv)
        target_compile_definitions(${target} PRIVATE NOVII_PV)
    endif()
    if(backend STREQUAL cloudpilot)
        target_compile_definitions(${target} PRIVATE OMNI_REMOTE EMULATION_LEVEL=EMULATION_UNIX)
    endif()
    target_compile_options(${target} PRIVATE -O3 -Wno-multichar -Wno-unused-parameter)
    target_link_libraries(${target} PRIVATE SDL2::SDL2 android log)
endforeach()
